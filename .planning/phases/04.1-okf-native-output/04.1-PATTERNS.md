### 1.1 Module declaration + error shape (the "new module in vault/" pattern)


Add `pub mod okf;` (alphabetical slot after `log`). The error contract for a validator-style module is fixed by `validate.rs` — collect every violation, one error at the end (`validate.rs:201-203`):

```rust

        return Err(CoreError::Validation(violations.join("; ")));
```

```rust
pub fn extract_provenance(body: &str) -> Vec<String> {
    let mut seen = HashSet::new();
    let mut out = Vec::new();
    let mut rest = body;
    while let Some(open) = rest.find("^[") {
        let after = &rest[open + 2..];
        let Some(close) = after.find(']') else { break };
        let target = after[..close].trim();
        if !target.is_empty() && seen.insert(target.to_string()) {
            out.push(target.to_string());
        }
        rest = &after[close + 1..];
    }
    out
}
```
